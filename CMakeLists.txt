cmake_minimum_required(VERSION 3.16)
project(lowlat_md_pipeline CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(NATIVE "Compile with -march=native" ON)
set(CMAKE_CXX_FLAGS_RELEASE "-O3")
add_compile_options(-Wall -Wextra)
if(NATIVE)
  add_compile_options(-march=native)
endif()
find_package(Threads REQUIRED)

add_library(alloc_counter OBJECT src/alloc_counter.cpp)
target_include_directories(alloc_counter PUBLIC include)

foreach(app gen replay bench_book bench_spsc)
  add_executable(${app} src/${app}.cpp $<TARGET_OBJECTS:alloc_counter>)
  target_include_directories(${app} PRIVATE include)
  target_link_libraries(${app} PRIVATE Threads::Threads)
endforeach()

enable_testing()
foreach(t test_parser test_book_diff test_hash test_spsc test_noalloc)
  add_executable(${t} tests/${t}.cpp $<TARGET_OBJECTS:alloc_counter>)
  target_include_directories(${t} PRIVATE include tests)
  target_link_libraries(${t} PRIVATE Threads::Threads)
  add_test(NAME ${t} COMMAND ${t})
endforeach()
